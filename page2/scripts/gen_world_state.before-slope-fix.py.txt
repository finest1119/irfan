# ---------------- terrain (analytic; the grid below is sampled from this) ----------------
RIDGE_Z = round(80 * FT, 2)          # 24.38 m
CREEK_TOP = (70.0, 5.42)             # (y, bed z) top of the steep section = hydro intake
CREEK_BOTTOM = (9.0, -2.50)          # (y, bed z) bottom = hydro turbine; fall 7.92 m = 26 ft over 61 m = 200 ft
FALL_Y = (36.0, 35.0)                # small waterfall: bed drops 2.0 m (6.5 ft) here
FALL_TOP_Z, FALL_BOT_Z = 2.40, 0.40
POND = (48.0, -60.0, 9.0, -3.40)     # x, y, radius, water surface z
RIDGE_CREST_X = 90.0
PAD = (96.0, 52.0, 7.0)              # ridge pad centre x, y, half-size


def creek_x(y):
    return 46.0 + 2.0 * math.sin(y / 14.0)


def bed(y):
    yt, zt = CREEK_TOP
    yb, zb = CREEK_BOTTOM
    if y >= yt:
        return zt + (y - yt) * 0.05
    if y >= FALL_Y[0]:
        return FALL_TOP_Z + (y - FALL_Y[0]) * (zt - FALL_TOP_Z) / (yt - FALL_Y[0])
    if y >= FALL_Y[1]:
        return FALL_BOT_Z + (y - FALL_Y[1]) * (FALL_TOP_Z - FALL_BOT_Z)
    if y >= yb:
        return zb + (y - yb) * (FALL_BOT_Z - zb) / (FALL_Y[1] - yb)
    return zb + (y - yb) * (0.9 / 64.0)          # gentle run to the pond (-3.4 at y=-55)


def smooth(t):
    t = max(0.0, min(1.0, t))
    return 0.5 * t + 0.5 * t * t * (3 - 2 * t)   # half linear, half smoothstep -> max grade 1.25x average


def west_base(x, y):
    return 0.0 if y <= 30 else (y - 30) * 0.06


def crest(y):
    return RIDGE_Z if y >= 20 else RIDGE_Z - (20 - y) * 0.06


def height(x, y):
    if x < -18:                                  # street side, flat
        return 0.0
    if math.hypot(x - POND[0], y - POND[1]) < POND[2]:
        return POND[3] - 1.2
    cx = creek_x(y)
    d = x - cx
    b = bed(y)
    if abs(d) <= 1.5:
        return b
    if d < 0:
        return b + 0.6 + (west_base(x, y) - (b + 0.6)) * smooth((-d - 1.5) / 10.5)
    return b + 0.6 + (crest(y) - (b + 0.6)) * smooth((d - 1.5) / (RIDGE_CREST_X - cx - 1.5))


def _solve_tank_x(y=52.0, target=FALL_BOT_Z + 65 * FT):
    lo, hi = creek_x(y) + 2, RIDGE_CREST_X
    for _ in range(40):
        mid = (lo + hi) / 2
        lo, hi = (mid, hi) if height(mid, y) < target else (lo, mid)
    return round(lo, 1)


TANK_ALT_X = _solve_tank_x()


def r(v, n=2):
    return [round(a, n) for a in v] if isinstance(v, (list, tuple)) else round(v, n)


def zone(pos, size, yaw=0, desc="", **kw):
    z = {"pos": r(pos), "size": r(size), "yaw": yaw, "desc": desc}
    z.update(kw)
    return z


def g(x, y, dz=0.0):
    return [round(x, 2), round(y, 2), round(height(x, y) + dz, 2)]
